public class GenresService(HttpClient client)
{
    public async Task<List<Genre>?> GetGenres()
    {
        var response = await client.GetFromJsonAsync<List<Genre>>("genres");
        return response;
    }

    public async Task AddGenre(Genre genre)
    {
        using var response = await client.PostAsJsonAsync("genres", genre);
        response.EnsureSuccessStatusCode();

        var content = await response.Content.ReadAsStringAsync();
        var options = new JsonSerializerOptions
        {
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase
        };
        genre = JsonSerializer.Deserialize<Genre>(content, options);

        genre = await response.Content.ReadFromJsonAsync<Genre>();
    }

    public async Task UpdateGenre(Genre genre)
    {
        var response = await client.PutAsJsonAsync($"genres/{genre.Id}", genre);
        if (response.IsSuccessStatusCode)
            Console.WriteLine("ok");
        else if (response.StatusCode == HttpStatusCode.BadRequest)
            throw new Exception("400");
        else if (response.StatusCode == HttpStatusCode.NotFound)
            Console.WriteLine("404");
    }
}
